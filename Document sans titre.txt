def afficher_repertoire():
    print(f"=== PROFIL: {nickel_data['profil']['nom_affichage']} ===")
    print(f"URL: {nickel_data['profil']['url_profil']}\n")
    
    print("--- LIENS EXTERNES ---")
    for lien in nickel_data["liens_externes"]:
        print(f"• {lien}")
    print("\n--- LISTE DES DÉPÔTS (26/26) ---")
    
    for idx, repo in enumerate(nickel_data["depots"], 1):
        statut = f"[{repo['visibilite']}]"
        langue = f" ({repo['langage']})" if repo['langage'] else ""
        fork = f" [Fork: {repo['fork_de']}]" if "fork_de" in repo else ""
        print(f"{idx:02d}. {repo['nom']} {statut}{langue}{fork}")
        if repo['description']:
            print(f"    Description: {repo['description']}")
        print(f"    URL: {repo['url']}")
        print(f"    Dernière mise à jour: {repo['mise_a_jour']}\n")


if __name__ == "__main__":
    afficher_repertoire()
