def treeofthought(etat0, profondeur, beam, rng):
    """Beam search sur un arbre de pensées. Chaque état a une valeur V(s).
    On garde les beam meilleurs à chaque niveau."""
    frontiere = [(etat0, 0.5)]                 # (état, valeur)
    trace = []
    for niveau in range(profondeur):
        candidats = []
        for etat, val in frontiere:
            for _ in range(3):                 # facteur de branchement = 3
                nouvel_etat = etat + rng.normal()*0.5
                # fonction d'évaluation V(s) : ici un paysage avec un optimum
                V = np.exp(-(nouvel_etat - 2.0)2)

        frontiere = candidats[:beam]           # garder les beam meilleurs
        trace.append(frontiere[0][1])
    return frontiere[0], trace
