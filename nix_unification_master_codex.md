    @staticmethod

        if limit < 2:
            return []
        sqrt_lim = int(math.isqrt(limit))
        segment_size = max(sqrt_lim, 32768)
        
        # Crible initial pour trouver les petits premiers
        is_prime_init = [True] * (sqrt_lim + 1)
        for p in range(2, int(math.isqrt(sqrt_lim)) + 1):
            if is_prime_init[p]:
                for i in range(p*p, sqrt_lim + 1, p):
                    is_prime_init[i] = False
        primes = [p for p in range(2, sqrt_lim + 1) if is_prime_init[p]]
        
        result = []
        low = sqrt_lim + 1
        high = low + segment_size - 1
        
        # Ajout des premiers initialisés
        result.extend([p for p in primes if p <= limit])
        
        is_prime = [True] * segment_size
        while low <= limit:
            if high > limit:
                high = limit
            is_prime[:] = [True] * len(is_prime)
            
            for p in primes:
                start = max(p * p, ((low + p - 1) // p) * p)
                for i in range(start - low, high - low + 1, p):
                    is_prime[i] = False
            
            for i in range(high - low + 1):
                if is_prime[i] and (low + i) > 1:
                    result.append(low + i)
            
            low += segment_size
            high += segment_size

    """

    def get_hilbert_dimension(self):
        """Retourne la dimension exacte de l'espace de Hilbert quantique de Willow (2^105)"""
        dim = 2 ** self.num_qubits
        log10_dim = self.num_qubits * math.log10(2)
        # Renvoie la mantisse et l'exposant pour l'affichage de notre gigantisme
        exponent = int(math.floor(log10_dim))
        mantissa = 10 ** (log10_dim - exponent)
        return dim, mantissa, exponent

    @staticmethod
    def transmon_hamiltonian(E_C=1.0, E_J=50.0, ng=0.0, N_cut=10):
        """
        Diagonalise l'Hamiltonien Josephson du qubit transmon de Willow :
        H = 4 * E_C * (n - ng)^2 - E_J * cos(phi)
        cos(phi) couple les états de charge |n> et |n +/- 1>
        """
        dim = 2 * N_cut + 1
        H = np.zeros((dim, dim))
        
        # Base des états de charge |n>
        states = list(range(-N_cut, N_cut + 1))
        
        # 1. Terme d'énergie de charge (diagonale)
        for i, n in enumerate(states):
            H[i, i] = 4.0 * E_C * ((n - ng)**2)
            
        # 2. Terme d'énergie Josephson (cos(phi) = 0.5 * (e^i_phi + e^-i_phi)) (couplages hors diagonale)
        for i in range(dim - 1):
            H[i, i + 1] = - 0.5 * E_J
            H[i + 1, i] = - 0.5 * E_J
            
        # Résolution spectrale de la quincaillerie
        eigenvalues, eigenvectors = np.linalg.eigh(H)
        
        # Énergies des niveaux fondamentaux
        E0 = eigenvalues[0]
        E1 = eigenvalues[1]
        E2 = eigenvalues[2]
        
        # Transitions et anharmonicité absolue
        transition_01 = E1 - E0
        transition_12 = E2 - E1
        anharmonicity = transition_12 - transition_01
        
        return eigenvalues[:5], transition_01, anharmonicity

    @staticmethod
    def surface_code_error_scaling(d, p_physical, p_threshold=0.01):
        """Loi d'échelle de décroissance logique de Willow sous Surface Code"""
        # epsilon_L = A * (p_physical / p_threshold) ^ ((d + 1) / 2)

        exponent = (d + 1) / 2.0
        ratio = p_physical / p_threshold
        logical_error = p_threshold * (ratio ** exponent)
        return float(logical_error)
